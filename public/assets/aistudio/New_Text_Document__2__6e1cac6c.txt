/**
 * ============================================================================
 * Web App Routing (GET & POST API สำหรับ Vercel และ Power Automate)
 * ============================================================================
 */
function doGet(e) {
  try {
    var params = (e && e.parameter) || {};
    var action = String(params.action || '').trim().toLowerCase();

    // 1. รองรับ Callback จาก Power Automate ผ่าน HTTP GET
    if (action === 'powerautomatecallback') {
      return handleAutomateCallback(params);
    }

    // 2. API สำหรับหน้าเว็บ Vercel (GET Requests)
    if (action === 'getapplications') {
      return jsonResponse(getApplicationsList(params.username, params.role));
    }

    if (action === 'getuserquota') {
      return jsonResponse(getUserQuota(params.username));
    }

    if (action === 'getallquota') {
      return jsonResponse(getAllQuotaOverview());
    }

    if (action === 'getemailpreview') {
      return jsonResponse(getEmailPreviewData(params.rowId));
    }

    // กรณีเปิด URL ตรงๆ
    return jsonResponse({
      success: true,
      message: "Mitr Phol Internship API is running.",
      timestamp: new Date().toISOString()
    });
  } catch (err) {
    return jsonResponse({
      success: false,
      message: "doGet Error: " + String(err)
    });
  }
}

function doPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) {
      return jsonResponse({
        success: false,
        message: "Missing POST body"
      });
    }

    var data = JSON.parse(e.postData.contents);
    var action = String(data.action || "").trim().toLowerCase();

    // 1. Callback จาก Power Automate
    if (action === "powerautomatecallback") {
      return handleAutomateCallback(data);
    }

    // 2. API สำหรับหน้าเว็บ Vercel (POST Requests)
    if (action === "login") {
      return jsonResponse(checkCompanyLogin(data.username, data.password));
    }

    if (action === "submitform") {
      return jsonResponse(processForm(data.formData));
    }

    if (action === "claim") {
      return jsonResponse(claimApplication(data.rowId, data.reviewerName));
    }

    if (action === "usercancelclaim") {
      return jsonResponse(userCancelClaim(data.rowId, data.username));
    }

    if (action === "adminreset") {
      return jsonResponse(adminResetStatus(data.rowId, data.adminUsername));
    }

    if (action === "sendadminemail") {
      return jsonResponse(sendAdminNotificationEmail(data.rowId, data.adminUsername));
    }

    if (action === "updatequota") {
      return jsonResponse(updateQuotaApproved(data.updatedList, data.adminName));
    }

    return jsonResponse({
      success: false,
      message: "Unknown action",
      action: action
    });

  } catch (err) {
    console.error("doPost Error: " + err);
    return jsonResponse({
      success: false,
      message: String(err)
    });
  }
}

/**
 * Helper JSON Response
 */
function jsonResponse(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

/**
 * ============================================================================
 * Power Automate Callback Handler
 * ============================================================================
 */
function handleAutomateCallback(params) {
  var lock = LockService.getScriptLock();

  try {
    var rowId = parseInt(params.sheetRow, 10);
    var outcome = String(params.outcome || "").trim().toLowerCase();
    var approverName = String(params.approverName || "").trim();
    var approverEmail = String(params.approverEmail || "").trim();
    var requesterEmail = String(params.requesterEmail || "").trim();
    var applicantName = String(params.applicantName || "นักศึกษาฝึกงาน").trim();

    if (!rowId || rowId < 2) {
      return jsonResponse({
        success: false,
        message: "Invalid sheetRow",
        sheetRow: params.sheetRow
      });
    }

    if (outcome !== "approve" && outcome !== "reject") {
      return jsonResponse({
        success: false,
        message: "Invalid outcome",
        outcome: params.outcome
      });
    }

    if (!approverName) {
      approverName = approverEmail || "Approver";
    }

    if (!lock.tryLock(10000)) {
      return jsonResponse({
        success: false,
        message: "System busy. Please retry later."
      });
    }

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("ข้อมูลผู้สมัคร") || ss.getSheetByName("Applications");

    if (!sheet) {
      return jsonResponse({
        success: false,
        message: "Application sheet not found"
      });
    }

    if (rowId > sheet.getLastRow()) {
      return jsonResponse({
        success: false,
        message: "sheetRow exceeds last row",
        sheetRow: rowId
      });
    }

    var isApproved = outcome === "approve";
    var statusValue = isApproved ? "reviewed" : "rejected";
    var statusText = isApproved ? "อนุมัติแล้ว" : "ไม่อนุมัติ";

    // Col U = สถานะ (21), Col W = วันที่พิจารณา (23)
    sheet.getRange(rowId, 21).setValue(statusValue);
    sheet.getRange(rowId, 23).setValue(new Date());

    SpreadsheetApp.flush();

    logActionHistory(
      "PowerAutomate_" + (isApproved ? "Approve" : "Reject"),
      rowId,
      approverName,
      "ผลพิจารณา: " + statusText + (approverEmail ? " | " + approverEmail : "")
    );

    // ส่งแจ้งเตือนผู้ยื่นเรื่อง (Requester)
    var emailSent = false;
    if (requesterEmail && requesterEmail.indexOf("@") > 0) {
      sendResultEmailToRequester(
        requesterEmail,
        applicantName,
        isApproved,
        approverName
      );
      emailSent = true;
    }

    return jsonResponse({
      success: true,
      message: "Approval result processed",
      sheetRow: rowId,
      outcome: isApproved ? "Approve" : "Reject",
      status: statusValue,
      approverName: approverName,
      requesterEmail: requesterEmail,
      emailAttempted: emailSent
    });

  } catch (err) {
    console.error("handleAutomateCallback Error: " + err);
    return jsonResponse({
      success: false,
      message: String(err),
      sheetRow: params.sheetRow || null
    });
  } finally {
    if (lock.hasLock()) {
      lock.releaseLock();
    }
  }
}

function getApplicationsList(username, role) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName('ข้อมูลผู้สมัคร') || ss.getSheetByName('Applications');
    if (!sheet || sheet.getLastRow() <= 1) return [];

    var curUser = String(username || '').trim().toLowerCase();
    var userRole = String(role || '').trim().toLowerCase();

    var isAdmin = (userRole === 'admin');
    if (!isAdmin && curUser) {
      isAdmin = (getUserRoleFromSheet(curUser) === 'admin');
    }

    var lastRow = sheet.getLastRow();
    var maxCols = Math.max(sheet.getLastColumn(), 27);
    var data = sheet.getRange(2, 1, lastRow - 1, maxCols).getValues();
    var result = [];

    var approverMap = getApproverDisplayMap();

    for (var i = 0; i < data.length; i++) {
      var row = data[i];
      if (!row[2] && !row[6] && !row[4]) continue;

      var rawStatus = String(row[20] || '').trim().toLowerCase();
      var status = (!rawStatus || rawStatus === 'unassigned') ? 'unassigned' : rawStatus;
      var reviewer = String(row[21] || '').trim();
      var reviewerNormalized = reviewer.toLowerCase();

      var isUnassigned = (status === 'unassigned' || !reviewer || reviewer === '-');
      var isMyItem = (reviewerNormalized === curUser);

      if (!isAdmin && curUser && !isUnassigned && !isMyItem) {
        continue;
      }

      var displayApprover = '-';
      if (!isUnassigned) {
        displayApprover = approverMap[reviewerNormalized] || reviewer;
      }

      var rawProfileUrl = String(row[18] || '').trim();
      var directProfileUrl = formatDriveImageUrl(rawProfileUrl);

      result.push({
        id: i + 1,
        sheetRow: i + 2,
        timestamp: (row[0] instanceof Date) ? Utilities.formatDate(row[0], Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss') : String(row[0] || ''),
        location: String(row[1] || ''),
        fullName: String(row[2] || ''),
        nickName: String(row[3] || ''),
        phone: String(row[4] || ''),
        lineId: String(row[5] || ''),
        emailStudent: String(row[6] || ''),
        emailInst: String(row[7] || ''),
        address: String(row[8] || ''),
        degree: String(row[9] || ''),
        institution: String(row[10] || ''),
        major: String(row[11] || ''),
        startDate: (row[12] instanceof Date) ? formatDate(row[12]) : String(row[12] || ''),
        endDate: (row[13] instanceof Date) ? formatDate(row[13]) : String(row[13] || ''),
        workUnit: String(row[14] || ''),
        skills: String(row[15] || ''),
        emergencyName: String(row[16] || ''),
        emergencyPhone: String(row[17] || ''),
        profileImageLink: directProfileUrl,
        fileLinks: String(row[19] || ''),
        reviewStatus: status,
        status: status,
        reviewerUsername: reviewer,
        approverDisplayName: displayApprover,
        reviewedAt: (row[22] instanceof Date) ? Utilities.formatDate(row[22], Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss') : String(row[22] || ''),
        selectedDept: String(row[23] || '-'),    // Col X
        selectedSection: String(row[24] || '-'),  // Col Y
        emailSentStatus: String(row[26] || '')    // Col AA (Index 26 = คอลัมน์ที่ 27)
      });
    }

    return result.reverse();
  } catch (error) {
    Logger.log('getApplicationsList Error: ' + error.toString());
    return { success: false, message: 'เกิดข้อผิดพลาดในการดึงข้อมูล: ' + error.message };
  }
}

function getApproverDisplayMap() {
  var map = {};
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName('User') || ss.getSheetByName('CompanyUsers');
    if (!sheet) return map;
    var data = sheet.getDataRange().getValues();

    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      var empId = String(row[0] || '').trim().toLowerCase();
      var userEmail = String(row[10] || '').trim().toLowerCase();
      var userPrefix = userEmail.split('@')[0];
      var approverName = String(row[11] || '').trim();
      var approverEmail = String(row[12] || '').trim().toLowerCase();

      if (approverName) {
        if (empId) map[empId] = approverName;
        if (userEmail) map[userEmail] = approverName;
        if (userPrefix) map[userPrefix] = approverName;
        if (approverEmail) map[approverEmail] = approverName;
      }
    }
  } catch(e) {}
  return map;
}

function claimApplication(rowId, reviewerName) {
  try {
    var rId = parseInt(rowId, 10);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var appSheet = ss.getSheetByName('ข้อมูลผู้สมัคร') || ss.getSheetByName('Applications');
    if (!appSheet) return { success: false, message: 'ไม่พบชีตข้อมูลผู้สมัคร' };

    var currentReviewer = appSheet.getRange(rId, 22).getValue();
    if (currentReviewer && String(currentReviewer).trim() !== '' && String(currentReviewer).trim() !== '-') {
      return { success: false, message: 'ใบสมัครนี้มีผู้รับเรื่องแล้ว: ' + currentReviewer };
    }

    // ข้อมูลจากชีต User
    var userSheet = ss.getSheetByName('User') || ss.getSheetByName('CompanyUsers');
    var approverName = '';
    var approverEmail = '';
    var requesterEmail = '';
    var userFullName = reviewerName;
    var userDept = '-';       // ฝ่าย (Col F ในชีต User)
    var userSection = '-';    // แผนก (Col I ในชีต User)
    var userAffiliation = '-';// สังกัด (Col Q ในชีต User - Index 16)

    if (userSheet) {
      var uData = userSheet.getDataRange().getValues();
      var cleanReviewer = String(reviewerName || '').trim().toLowerCase();

      for (var i = 1; i < uData.length; i++) {
        var row = uData[i];
        var empId = String(row[0] || '').trim().toLowerCase();
        var dbEmail = String(row[10] || '').trim().toLowerCase();
        var emailPrefix = dbEmail.split('@')[0];

        if (cleanReviewer === empId || cleanReviewer === dbEmail || cleanReviewer === emailPrefix) {
          var title = String(row[1] || '').trim();
          var fName = String(row[2] || '').trim();
          var lName = String(row[3] || '').trim();
          if (fName || lName) userFullName = (title + ' ' + fName + ' ' + lName).trim();

          userDept = String(row[5] || '-').trim();        // Col F (index 5) = ฝ่าย
          userSection = String(row[8] || '-').trim();     // Col I (index 8) = แผนก
          userAffiliation = String(row[16] || '-').trim(); // Col Q (index 16) = สังกัด

          requesterEmail = String(row[10] || '').trim();
          approverName = String(row[11] || '').trim();
          approverEmail = String(row[12] || '').trim();
          break;
        }
      }
    }

    // บันทึกสถานะ และ ผู้รับเรื่อง
    appSheet.getRange(rId, 21).setValue('pending');
    appSheet.getRange(rId, 22).setValue(reviewerName);
    appSheet.getRange(rId, 23).setValue(new Date());

    // บันทึก ฝ่าย (Col X = 24), แผนก (Col Y = 25), และ สังกัด (Col Z = 26)
    appSheet.getRange(rId, 24).setValue(userDept);
    appSheet.getRange(rId, 25).setValue(userSection);
    appSheet.getRange(rId, 26).setValue(userAffiliation);

    logActionHistory('Claim', rId, reviewerName, 'รับเรื่องรอการพิจารณา (สังกัด: ' + userAffiliation + ')');

    var applicantName = String(appSheet.getRange(rId, 3).getValue() || '');
    var institution = String(appSheet.getRange(rId, 11).getValue() || '');
    var major = String(appSheet.getRange(rId, 12).getValue() || '');
    var workUnit = String(appSheet.getRange(rId, 15).getValue() || '');
    var rawFileLinks = String(appSheet.getRange(rId, 20).getValue() || '').trim(); // Col T

    // ดึงไฟล์แนบจริงจาก Google Drive แปลงเป็น Base64
    var attachmentsList = [];
    if (rawFileLinks) {
      var lines = rawFileLinks.split('\n');
      for (var k = 0; k < lines.length; k++) {
        var line = lines[k].trim();
        if (!line) continue;
        
        var fileName = 'เอกสารแนบ_' + (k + 1);
        var fileUrl = line;
        
        var parts = line.split(': http');
        if (parts.length >= 2) {
          fileName = parts[0].trim();
          fileUrl = 'http' + parts.slice(1).join(': http').trim();
        }

        // ดึง File ID จาก URL Google Drive
        var fileId = '';
        var matchD = fileUrl.match(/\/d\/([a-zA-Z0-9_-]+)/);
        if (matchD && matchD[1]) {
          fileId = matchD[1];
        } else {
          var matchId = fileUrl.match(/id=([a-zA-Z0-9_-]+)/);
          if (matchId && matchId[1]) fileId = matchId[1];
        }

        if (fileId) {
          try {
            var driveFile = DriveApp.getFileById(fileId);
            var blob = driveFile.getBlob();
            var realName = driveFile.getName();
            var ext = realName.indexOf('.') !== -1 ? realName.substring(realName.lastIndexOf('.')) : '';
            var finalName = fileName.indexOf('.') !== -1 ? fileName : (fileName + ext);

            attachmentsList.push({
              name: finalName,
              content: Utilities.base64Encode(blob.getBytes())
            });
          } catch(errDrive) {
            Logger.log('ไม่สามารถดึงไฟล์แนบ ID ' + fileId + ': ' + errDrive.toString());
          }
        }
      }
    }

    var automateStatus = '';
    if (POWER_AUTOMATE_WEBHOOK_URL && POWER_AUTOMATE_WEBHOOK_URL.indexOf('http') === 0) {
      var payload = {
        action: "claimApprovalRequest",
        sheetRow: rId,
        applicantName: applicantName,
        institution: institution,
        major: major,
        workUnit: workUnit,
        approverName: approverName,
        approverEmail: approverEmail,
        requesterName: userFullName,
        requesterEmail: requesterEmail,
        requesterDepartment: userDept,
        requesterSection: userSection,
        attachments: attachmentsList,
        webAppUrl: getWebAppUrl()
      };

      try {
        var options = {
          method: 'post',
          contentType: 'application/json',
          payload: JSON.stringify(payload),
          muteHttpExceptions: true
        };
        UrlFetchApp.fetch(POWER_AUTOMATE_WEBHOOK_URL, options);
        automateStatus = ' (ส่งคำขออนุมัติไปยัง ' + (approverName || approverEmail) + ' สำเร็จ)';
      } catch (err) {
        Logger.log('Power Automate Error: ' + err.toString());
        automateStatus = ' (เชื่อมต่อ Automate ขัดข้อง: ' + err.message + ')';
      }
    }

    return { 
      success: true, 
      message: 'รับ นศ. ฝึกงาน เรียบร้อยแล้ว' + automateStatus,
      approverName: approverName
    };
  } catch (error) {
    return { success: false, message: 'เกิดข้อผิดพลาด: ' + error.toString() };
  }
}

function userCancelClaim(rowId, currentUsername) {
  try {
    var rId = parseInt(rowId, 10);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName('ข้อมูลผู้สมัคร') || ss.getSheetByName('Applications');
    if (!sheet) return { success: false, message: 'ไม่พบชีตข้อมูล' };

    var reviewer = String(sheet.getRange(rId, 22).getValue() || '').trim().toLowerCase();
    var cleanUser = String(currentUsername || '').trim().toLowerCase();

    if (reviewer !== cleanUser) {
      return { success: false, message: 'คุณไม่มีสิทธิ์ยกเลิกรายการที่ไม่ได้เป็นผู้รับเรื่อง' };
    }

    sheet.getRange(rId, 21).setValue('unassigned');
    sheet.getRange(rId, 22).setValue('-');
    sheet.getRange(rId, 23).setValue('');
    sheet.getRange(rId, 24).setValue('-'); // ล้าง ฝ่าย
    sheet.getRange(rId, 25).setValue('-'); // ล้าง แผนก
    sheet.getRange(rId, 26).setValue('-'); // ล้าง สังกัด (Col Z)

    logActionHistory('UserCancelClaim', rId, currentUsername, 'ยกเลิกการรับเรื่อง คืนเป็น unassigned');

function saveBase64File(folder, fileName, base64Data) {
  var parts = base64Data.split(',');
  var contentType = parts[0].split(':')[1].split(';')[0];
  var decodedData = Utilities.base64Decode(parts[1]);
  var blob = Utilities.newBlob(decodedData, contentType, fileName);
  var file = folder.createFile(blob);
  file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  return file;
}

}

function formatDriveImageUrl(url) {
  if (!url) return '';
  var urlStr = String(url);
  var fileId = '';
  var matchD = urlStr.match(/\/d\/([a-zA-Z0-9_-]+)/);
  if (matchD && matchD[1]) fileId = matchD[1];
  else {
    var matchId = urlStr.match(/id=([a-zA-Z0-9_-]+)/);
    if (matchId && matchId[1]) fileId = matchId[1];
    else {
      var matchAny = urlStr.match(/[-\w]{25,}/);
      if (matchAny) fileId = matchAny[0];
    }
  }
  return fileId ? ('https://lh3.googleusercontent.com/d/' + fileId) : urlStr;
}

function formatDate(dateObj) {
  if (!dateObj) return '-';
  if (!(dateObj instanceof Date)) {
    var d = new Date(dateObj);
    if (isNaN(d.getTime())) return String(dateObj);
    dateObj = d;
  }
  var day = String(dateObj.getDate()).padStart(2, '0');
  var month = String(dateObj.getMonth() + 1).padStart(2, '0');
  var year = dateObj.getFullYear();
  return day + '/' + month + '/' + year;
}

 * Quota Management Functions
 * ============================================================================
 */
function getUserQuota(username) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var userSheet = ss.getSheetByName('User') || ss.getSheetByName('CompanyUsers');
    var quotaSheet = ss.getSheetByName('Quota');
    
    var defaultResult = { affiliation: '-', approved: 0, actual: 0, remaining: 0 };
    if (!userSheet || !quotaSheet) return defaultResult;

    var cleanUser = String(username || '').trim().toLowerCase();
    var cleanNormalized = cleanUser.replace(/[\._-]/g, '');
    var uData = userSheet.getDataRange().getValues();
    var userAffiliation = '';

    for (var i = 1; i < uData.length; i++) {
      var row = uData[i];
      var empId = String(row[0] || '').trim().toLowerCase();
      var dbEmail = String(row[10] || '').trim().toLowerCase();
      var emailPrefix = dbEmail.split('@')[0];
      var normalizedPrefix = emailPrefix.replace(/[\._-]/g, '');

      if (
        cleanUser === dbEmail ||
        cleanUser === emailPrefix ||
        cleanNormalized === normalizedPrefix ||
        cleanUser === empId
      ) {
        userAffiliation = String(row[16] || '').trim(); // Col Q = สังกัด
        break;
      }
    }

    if (!userAffiliation || userAffiliation === '-') {
      return defaultResult;
    }

    var qData = quotaSheet.getDataRange().getValues();
    for (var j = 1; j < qData.length; j++) {
      var qRow = qData[j];
      var qAffiliation = String(qRow[1] || '').trim(); // Col B = สังกัด

      if (qAffiliation && (qAffiliation === userAffiliation || userAffiliation.indexOf(qAffiliation) !== -1 || qAffiliation.indexOf(userAffiliation) !== -1)) {
        return {
          affiliation: userAffiliation,
          approved: Number(qRow[2]) || 0,   // Col C = อนุมัติ
          actual: Number(qRow[3]) || 0,     // Col D = เกิดจริง
          remaining: Number(qRow[4]) || 0   // Col E = คงเหลือ
        };
      }
    }

    return { affiliation: userAffiliation, approved: 0, actual: 0, remaining: 0 };
  } catch (err) {
    Logger.log('getUserQuota Error: ' + err.toString());
    return { affiliation: '-', approved: 0, actual: 0, remaining: 0 };
  }
}

    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName('Quota');
    if (!sheet) return { success: false, message: 'ไม่พบชีต Quota' };

    var data = sheet.getDataRange().getValues();
    var list = [];
    var totalApproved = 0;
    var totalActual = 0;
    var totalRemaining = 0;

    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      var name = String(row[1] || '').trim(); // Col B: สังกัด
      if (!name || name === 'รวม' || name.indexOf('รวม') !== -1) continue;

      var approved = Number(row[2]) || 0;  // Col C: อนุมัติ
      var actual = Number(row[3]) || 0;    // Col D: เกิดจริง
      var remaining = Number(row[4]) || 0; // Col E: คงเหลือ

      totalApproved += approved;
      totalActual += actual;
      totalRemaining += remaining;

      list.push({
        rowNum: i + 1,
        affiliation: name,
        approved: approved,
        actual: actual,
        remaining: remaining
      });
    }

    return {
      success: true,
      data: list,
      totals: {
        approved: totalApproved,
        actual: totalActual,
        remaining: totalRemaining
      }
    };
  } catch (err) {
    return { success: false, message: err.toString() };
  }
}

function updateQuotaApproved(updatedList, adminName) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName('Quota');
    if (!sheet) return { success: false, message: 'ไม่พบชีต Quota' };

    for (var i = 0; i < updatedList.length; i++) {
      var item = updatedList[i];
      var row = parseInt(item.rowNum, 10);
      var newVal = Number(item.approved) || 0;
      if (row > 1) {
        sheet.getRange(row, 3).setValue(newVal); // บันทึกลง Col C
      }
    }
    SpreadsheetApp.flush();

    logActionHistory('UpdateQuota', '-', adminName || 'Admin', 'แก้ไขโควต้าอนุมัติภาพรวม');
    return { success: true, message: 'บันทึกการแก้ไขโควต้าเรียบร้อยแล้ว' };
  } catch (err) {
    return { success: false, message: err.toString() };
  }
}
